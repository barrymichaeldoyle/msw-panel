import PackageManagerTabs from "../../components/PackageManagerTabs.astro";

Add two packages to your existing MSW setup:

<PackageManagerTabs packageName="msw-panel" dev />

```tsx
import { createMswPanelController } from "msw-panel";
import { MswPanel } from "msw-panel/react";

const controller = createMswPanelController({ runtime: worker });

function App() {
  return (
    <>
      <Routes />
      <MswPanel controller={controller} />
    </>
  );
}
```

A floating button appears in the corner. Click it to open the panel and toggle
any handler on or off. [Vue adapter (#4)](https://github.com/barrymichaeldoyle/msw-panel/issues/4)
and [Svelte adapter (#5)](https://github.com/barrymichaeldoyle/msw-panel/issues/5)
are the planned next first-party adapters. [Solid adapter (#6)](https://github.com/barrymichaeldoyle/msw-panel/issues/6)
and [Angular adapter (#7)](https://github.com/barrymichaeldoyle/msw-panel/issues/7)
are under consideration after that. Reactions and comments on those issues help
determine priority.
